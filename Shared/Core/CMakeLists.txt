cmake_minimum_required(VERSION 3.22)
project(MiniVoiceCore LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(minivoice_core SHARED Sources/MiniVoiceCore/core.cpp Sources/MiniVoiceCore/vendor/utf8proc.c)
target_include_directories(minivoice_core PRIVATE Sources/MiniVoiceCore)
target_compile_definitions(minivoice_core PRIVATE UTF8PROC_STATIC)
if(ANDROID)
    target_sources(minivoice_core PRIVATE Adapters/android_jni.cpp)
    target_link_options(minivoice_core PRIVATE "-Wl,-z,max-page-size=16384")
endif()
